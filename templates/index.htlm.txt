<script>
    const imageInput = document.getElementById("imageInput");
    const preview = document.getElementById("preview");
    const analyseButton = document.getElementById("analyseButton");
    const status = document.getElementById("status");

    let selectedFile = null;

    imageInput.addEventListener("change", function () {

        selectedFile = this.files[0];

        if (!selectedFile) {
            return;
        }

        const allowedTypes = [
            "image/jpeg",
            "image/png",
            "image/tiff"
        ];

        const extension = selectedFile.name.split(".").pop().toLowerCase();

        if (!allowedTypes.includes(selectedFile.type) &&
            !["jpg", "jpeg", "png", "tif", "tiff"].includes(extension)) {

            status.textContent = "Format d'image non accepté.";
            selectedFile = null;
            analyseButton.disabled = true;
            return;
        }

        preview.src = URL.createObjectURL(selectedFile);
        preview.classList.remove("hidden");

        analyseButton.disabled = false;
        status.textContent = "Image prête pour l'analyse.";
    });


    analyseButton.addEventListener("click", async function () {

        if (!selectedFile) {
            return;
        }

        const formData = new FormData();
        formData.append("image", selectedFile);

        analyseButton.disabled = true;
        status.textContent = "Analyse en cours... Veuillez patienter.";

        document.getElementById("results").classList.add("hidden");

        try {

            const response = await fetch("/analyser", {
                method: "POST",
                body: formData
            });

            const data = await response.json();

            if (!response.ok) {
                throw new Error(data.erreur || "Erreur pendant l'analyse.");
            }

            document.getElementById("results").classList.remove("hidden");

            document.getElementById("aircraftCount").textContent =
                data.nombre_avions;

            const resultImage = document.getElementById("resultImage");

            resultImage.src = data.image_annotee + "?t=" + Date.now();

            document.getElementById("downloadImage").href =
                data.image_annotee;

            document.getElementById("downloadJson").href =
                data.fichier_json;

            document.getElementById("downloadImage").download =
                "image_annotee.jpg";

            document.getElementById("downloadJson").download =
                "resultats.json";

            const aircraftList = document.getElementById("aircraftList");

            aircraftList.innerHTML = "";

            if (data.avions.length === 0) {

                aircraftList.textContent = "Aucun avion détecté.";

            } else {

                data.avions.forEach((avion, index) => {

                    const item = document.createElement("div");

                    const c = avion.coordinates;

                    item.textContent =
                        "Avion " + (index + 1) +
                        " — Classe : " + avion.class +
                        " — Coordonnées : (" +
                        c.xmin + ", " + c.ymin + ", " +
                        c.xmax + ", " + c.ymax + ")";

                    aircraftList.appendChild(item);
                });
            }

            status.textContent = "Analyse terminée avec succès !";

        } catch (error) {

            status.textContent = "Erreur : " + error.message;

        } finally {

            analyseButton.disabled = false;
        }
    });
</script>
